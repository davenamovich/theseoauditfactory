    84→export function parseRobots(text: string): RobotsInfo {
    85→  const sitemaps: string[] = [];
    86→  const lines = text.split(/\r?\n/).map((l) => l.trim());
    87→  for (const line of lines) {
    88→    const m = line.match(/^sitemap:\s*(\S+)/i);
    89→    if (m) sitemaps.push(m[1]);
    90→  }
    91→
    92→  // Build user-agent groups per RFC 9309: rules belong to the nearest
    93→  // preceding user-agent line(s); consecutive UA lines share one group.
    94→  interface Group {
    95→    agents: string[];
    96→    disallows: string[];
    97→    allows: string[];
    98→  }
    99→  const groups: Group[] = [];
   100→  let current: Group | null = null;
   101→  for (const line of lines) {
   102→    if (!line || line.startsWith('#')) continue;
   103→    const ua = line.match(/^user-agent:\s*(.*)$/i);
   104→    if (ua) {
   105→      const agent = ua[1].trim().toLowerCase();
   106→      if (current && current.disallows.length === 0 && current.allows.length === 0) {
   107→        current.agents.push(agent); // consecutive UA lines share a group
   108→      } else {
   109→        current = { agents: [agent], disallows: [], allows: [] };
   110→        groups.push(current);
   111→      }
   112→      continue;
   113→    }
   114→    const dis = line.match(/^disallow:\s*(.*)$/i);
   115→    if (dis && current) {
   116→      current.disallows.push(dis[1].trim());
   117→      continue;
   118→    }
   119→    const allow = line.match(/^allow:\s*(.*)$/i);
   120→    if (allow && current) current.allows.push(allow[1].trim());
   121→  }
   122→
   123→  const aiCrawlers: AiCrawlerPolicy[] = AI_CRAWLERS.map((name) => {
   124→    const lower = name.toLowerCase();
   125→    const group = groups.find((g) => g.agents.includes(lower));
   126→    if (!group) return { name, policy: 'not-mentioned' as const };
   127→    const blockedAll = group.disallows.some((d) => d === '/');
   128→    const hasSpecificRules = group.disallows.length > 0 || group.allows.length > 0;
   129→    if (blockedAll) return { name, policy: 'blocked' as const, detail: 'Disallow: / in robots.txt' };
   130→    if (hasSpecificRules) return { name, policy: 'allowed' as const, detail: 'Has specific rules (partial access)' };
   131→    return { name, policy: 'not-mentioned' as const };
   132→  });
   133→  return { found: true, text, sitemaps, aiCrawlers };
   134→}

   136→function finding(
   137→  code: string,
   138→  category: Finding['category'],
   139→  severity: Finding['severity'],
   140→  title: string,
   141→  evidence: string[],
   142→  explanation: string,
   143→  action: string,
   144→  impact: string,
   145→  effort: Finding['effort']
   146→): Finding {
   147→  return { code, category, severity, title, evidence, explanation, action, impact, effort };
   148→}

   153→export function analyzeTechnical(ctx: AnalyzeContext): AnalyzerOutput {
   154→  const checks: CheckResult[] = [];
   155→  const findings: Finding[] = [];
   156→  const p = ctx.homepage;
   157→
   158→  const https = ctx.finalUrl.startsWith('https://');
   159→  checks.push({
   160→    id: 'https',
   161→    label: 'HTTPS secure connection',
   162→    status: https ? 'pass' : 'fail',
   163→    detail: https ? 'Site served over HTTPS' : 'Site served over plain HTTP',
   164→  });
   165→  if (!https)
   166→    findings.push(
   167→      finding(
   168→        'no-https',
   169→        'technical',
   170→        'critical',
   171→        'Website is not served over HTTPS',
   172→        [`Final URL after redirects: ${ctx.finalUrl}`],
   173→        'Browsers label HTTP sites "Not secure," and Google uses HTTPS as a ranking signal. Many customers will not fill in a form on an unsecured page.',
   174→        'Install a TLS certificate (free with Let\u2019s Encrypt on most hosts) and redirect all HTTP traffic to HTTPS.',
   175→        'Trust, conversions, and rankings',
   176→        'quick'
   177→      )
   178→    );
   179→
   180→  const hops = ctx.redirectChain.length;
   181→  checks.push({
   182→    id: 'redirect-chain',
   183→    label: 'Redirect chain',
   184→    status: hops === 0 ? 'pass' : hops <= 2 ? 'pass' : hops <= 3 ? 'warn' : 'fail',
   185→    detail: hops === 0 ? 'Loads directly, no redirects' : `${hops} redirect hop(s) before the final page`,
   186→  });
   187→  if (hops >= 3)
   188→    findings.push(
   189→      finding(
   190→        'redirect-chain',
   191→        'technical',
   192→        'medium',
   193→        `Long redirect chain (${hops} hops)`,
   194→        [...ctx.redirectChain, ctx.finalUrl],
   195→        'Every extra redirect adds delay for visitors and wastes crawl budget for search engines.',
   196→        'Point links and redirects at the final URL directly so a single hop gets there.',
   197→        'Speed and crawl efficiency',
   198→        'quick'
   199→      )
   200→    );
   201→
   202→  checks.push({
   203→    id: 'robots-txt',
   204→    label: 'robots.txt present',
   205→    status: ctx.robots.found ? 'pass' : 'warn',
   206→    detail: ctx.robots.found ? 'Found at /robots.txt' : 'No robots.txt found (servers usually default to allow-all)',
   207→  });
   208→  if (!ctx.robots.found)
   209→    findings.push(
   210→      finding(
   211→        'no-robots',
   212→        'technical',
   213→        'low',
   214→        'No robots.txt file',
   215→        ['Request to /robots.txt returned no valid file'],
   216→        'robots.txt tells search engines (and AI crawlers) what to crawl. Its absence is usually safe but removes your control.',
   217→        'Add a simple robots.txt that references your sitemap. See the Technical SEO Checklist in our resource center.',
   218→        'Crawl control',
   219→        'quick'
   220→      )
   221→    );
   222→
   223→  checks.push({
   224→    id: 'robots-sitemap',
   225→    label: 'Sitemap declared in robots.txt',
   226→    status: ctx.robots.sitemaps.length > 0 ? 'pass' : 'warn',
   227→    detail: ctx.robots.sitemaps.length ? ctx.robots.sitemaps.slice(0, 2).join(', ') : 'No Sitemap: line in robots.txt',
   228→  });
   229→
   230→  checks.push({
   231→    id: 'xml-sitemap',
   232→    label: 'XML sitemap exists',
   233→    status: ctx.sitemap.found && (ctx.sitemap.urlCount ?? 0) > 0 ? 'pass' : ctx.sitemap.found ? 'pass' : 'fail',
   234→    detail:
   235→      ctx.sitemap.found && ctx.sitemap.urlCount
   236→        ? `${ctx.sitemap.urlCount} URLs listed`
   237→        : ctx.sitemap.found
   238→          ? ctx.sitemap.isIndex
   239→            ? 'Sitemap index found (references child sitemaps)'
   240→            : 'Sitemap found'
   241→          : 'No sitemap found at common locations or in robots.txt',
   242→  });
   243→  if (!ctx.sitemap.found || (ctx.sitemap.urlCount ?? 0) === 0)
   244→    findings.push(
   245→      finding(
   246→        'no-sitemap',
   247→        'technical',
   248→        'high',
   249→        'XML sitemap missing or empty',
   250→        ['No valid sitemap.xml found at /sitemap.xml or declared in robots.txt'],
   251→        'A sitemap is how you hand Google a complete list of your pages. Without one, new and important pages can be discovered late or never.',
   252→        'Generate a sitemap (most CMSs and plugins do this with one toggle) and reference it in robots.txt and Search Console.',
   253→        'Faster, more complete indexing',
   254→        'quick'
   255→      )
   256→    );
   257→
   258→  const noindex = /noindex/i.test(p.metaRobots);
   259→  checks.push({
   260→    id: 'indexable',
   261→    label: 'Homepage indexable',
   262→    status: noindex ? 'fail' : 'pass',
   263→    detail: noindex ? 'meta robots contains noindex' : 'No noindex directive found',
   264→  });
   265→  if (noindex)
   266→    findings.push(
   267→      finding(
   268→        'noindex',
   269→        'technical',
   270→        'critical',
   271→        'Homepage is marked noindex',
   272→        [`<meta name="robots" content="${p.metaRobots}">`],
   273→        'A noindex tag asks Google to keep this page out of search results entirely. On a homepage this is usually a staging setting that leaked to production.',
   274→        'Remove noindex from the homepage meta robots tag and request indexing in Google Search Console.',
   275→        'Entire search visibility',
   276→        'quick'
   277→      )
   278→    );
   279→
   280→  checks.push({
   281→    id: 'canonical',
   282→    label: 'Canonical tag',
   283→    status: p.canonical ? 'pass' : 'warn',
   284→    detail: p.canonical ? p.canonical : 'No rel=canonical on homepage',
   285→  });
   286→  if (!p.canonical)
   287→    findings.push(
   288→      finding(
   289→        'no-canonical',
   290→        'technical',
   291→        'low',
   292→        'No canonical tag on the homepage',
   293→        ['No <link rel="canonical"> found'],
   294→        'Canonical tags prevent duplicate-URL confusion (e.g., http/https, with/without trailing slash). Most templates include one; yours does not.',
   295→        'Add a self-referencing canonical link tag to the homepage template.',
   296→        'Duplicate-content hygiene',
   297→        'quick'
   298→      )
   299→    );
   300→
   301→  checks.push({
   302→    id: 'viewport',
   303→    label: 'Mobile viewport tag',
   304→    status: p.viewport ? 'pass' : 'fail',
   305→    detail: p.viewport ?? 'No <meta name="viewport"> found',
   306→  });
   307→  if (!p.viewport)
   308→    findings.push(
   309→      finding(
   310→        'no-viewport',
   311→        'technical',
   312→        'high',
   313→        'Page is not mobile-ready (no viewport tag)',
   314→        ['Missing <meta name="viewport"> tag'],
   315→        'Without a viewport tag, mobile browsers render the desktop layout zoomed out. Google indexes mobile-first, so this hurts every ranking.',
   316→        'Add <meta name="viewport" content="width=device-width, initial-scale=1"> to the page template.',
   317→        'All mobile rankings',
   318→        'quick'
   319→      )
   320→    );
   321→
   322→  const securityHeaders = ['strict-transport-security', 'x-content-type-options', 'x-frame-options', 'content-security-policy'].filter(
   323→    (h) => ctx.headers[h]
   324→  );
   325→  checks.push({
   326→    id: 'security-headers',
   327→    label: 'Security headers',
   328→    status: securityHeaders.length >= 3 ? 'pass' : securityHeaders.length >= 1 ? 'warn' : 'warn',
   329→    detail:
   330→      securityHeaders.length > 0
   331→        ? `Found: ${securityHeaders.join(', ')}`
   332→        : 'No HSTS, X-Content-Type-Options, X-Frame-Options or CSP headers found',
   333→  });
   334→
   335→  const brokenInternal = ctx.linkChecks.filter((l) => l.internal && !l.ok);
   336→  const brokenExternal = ctx.linkChecks.filter((l) => !l.internal && !l.ok);
   337→  const sampledCount = ctx.linkChecks.length;
   338→  const brokenStatus: CheckResult['status'] =
   339→    brokenInternal.length === 0 ? 'pass' : brokenInternal.length <= 2 ? 'warn' : 'fail';
   340→  checks.push({
   341→    id: 'broken-links',
   342→    label: `Broken links (sample of ${sampledCount})`,
   343→    status: sampledCount < 4 ? 'na' : brokenStatus,
   344→    detail:
   345→      sampledCount < 4
   346→        ? 'Not enough links found to test'
   347→        : `${brokenInternal.length} internal and ${brokenExternal.length} external broken of ${sampledCount} tested`,
   348→  });
   349→  if (sampledCount >= 4 && brokenInternal.length > 0)
   350→    findings.push(
   351→      finding(
   352→        'broken-links',
   353→        'technical',
   354→        brokenInternal.length > 2 ? 'high' : 'medium',
   355→        `${brokenInternal.length} broken internal link(s) found`,
   356→        brokenInternal.slice(0, 5).map((l) => `${l.url} returned ${l.status ?? 'an error'}`),
   357→        'Dead links frustrate customers and waste crawl budget. They also signal neglect to search engines.',
   358→        'Fix or remove the broken links listed in the evidence.',
   359→        'User trust and crawl efficiency',
   360→        'quick'
   361→      )
   362→    );
   363→
   364→  const encoding = ctx.headers['content-encoding'] ?? null;
   365→  checks.push({
   366→    id: 'compression',
   367→    label: 'Text compression (gzip/brotli)',
   368→    status: encoding ? 'pass' : 'warn',
   369→    detail: encoding ? `Served with ${encoding}` : 'No Content-Encoding header on HTML response',
   370→  });
   371→  if (!encoding)
   372→    findings.push(
   373→      finding(
   374→        'no-compression',
   375→        'technical',
   376→        'medium',
   377→        'HTML served without compression',
   378→        ['No Content-Encoding header on the homepage response'],
   379→        'Uncompressed HTML can be 3-5x larger than it needs to be, slowing every visit.',
   380→        'Enable gzip or brotli compression in your server or CDN config.',
   381→        'Load time on every page',
   382→        'quick'
   383→      )
   384→    );
   385→
   386→  checks.push({ id: 'favicon', label: 'Favicon', status: p.favicon ? 'pass' : 'warn', detail: p.favicon ? 'Icon link found' : 'No favicon link found' });
   387→  checks.push({
   388→    id: 'lang',
   389→    label: 'Language declared',
   390→    status: p.lang ? 'pass' : 'warn',
   391→    detail: p.lang ? `html lang="${p.lang}"` : 'No lang attribute on <html>',
   392→  });
   393→
   394→  return { key: 'technical', label: 'Technical SEO', checks, findings };
   395→}

   400→export function analyzeContent(ctx: AnalyzeContext): AnalyzerOutput {
   401→  const checks: CheckResult[] = [];
   402→  const findings: Finding[] = [];
   403→  const p = ctx.homepage;
   404→
   405→  const title = p.title;
   406→  const titleLen = title.length;
   407→  checks.push({
   408→    id: 'title',
   409→    label: 'Title tag',
   410→    status: titleLen === 0 ? 'fail' : titleLen >= 15 && titleLen <= 65 ? 'pass' : 'warn',
   411→    detail: titleLen === 0 ? 'Missing <title>' : `${titleLen} characters: "${title}"`,
   412→  });
   413→  if (titleLen === 0)
   414→    findings.push(
   415→      finding(
   416→        'no-title',
   417→        'content',
   418→        'critical',
   419→        'Homepage has no title tag',
   420→        ['<title> element is empty or missing'],
   421→        'The title tag is still the single strongest on-page signal and is what users click in search results.',
   422→        'Write a 50-60 character title: what you do + where you do it + brand.',
   423→        'Rankings and click-through rate',
   424→        'quick'
   425→      )
   426→    );
   427→  else if (titleLen < 15 || titleLen > 65)
   428→    findings.push(
   429→      finding(
   430→        'title-length',
   431→        'content',
   432→        'medium',
   433→        titleLen < 15 ? 'Title tag is too short' : 'Title tag is too long',
   434→        [`Current title (${titleLen} chars): "${title}"`],
   435→        titleLen < 15
   436→          ? 'Short titles waste the most valuable on-page space you have.'
   437→          : 'Long titles get cut off in search results, hiding your differentiators.',
   438→        titleLen < 15
   439→          ? 'Expand it to describe your main service and location.'
   440→          : 'Tighten it to under about 60 characters with the important words first.',
   441→        'Click-through rate and keyword relevance',
   442→        'quick'
   443→      )
   444→    );
   445→
   446→  const descLen = p.metaDescription.length;
   447→  checks.push({
   448→    id: 'meta-description',
   449→    label: 'Meta description',
   450→    status: descLen === 0 ? 'warn' : descLen >= 50 && descLen <= 165 ? 'pass' : 'warn',
   451→    detail: descLen === 0 ? 'Missing meta description' : `${descLen} characters`,
   452→  });
   453→  if (descLen === 0)
   454→    findings.push(
   455→      finding(
   456→        'no-meta-description',
   457→        'content',
   458→        'medium',
   459→        'Homepage has no meta description',
   460→        ['<meta name="description"> is missing'],
   461→        'Without a description, Google invents its own snippet from page text, which rarely sells your business as well as you could.',
   462→        'Write a 140-160 character description with your main service, area, and a reason to click.',
   463→        'Click-through rate',
   464→        'quick'
   465→      )
   466→    );
   467→
   468→  checks.push({
   469→    id: 'h1',
   470→    label: 'H1 heading',
   471→    status: p.h1s.length === 1 ? 'pass' : p.h1s.length === 0 ? 'fail' : 'warn',
   472→    detail: p.h1s.length === 0 ? 'No H1 found' : `${p.h1s.length} H1(s): ${p.h1s.slice(0, 2).join(' | ')}`,
   473→  });
   474→  if (p.h1s.length === 0)
   475→    findings.push(
   476→      finding(
   477→        'no-h1',
   478→        'content',
   479→        'high',
   480→        'Homepage has no H1 heading',
   481→        ['Zero <h1> elements found'],
   482→        'The H1 tells both visitors and search engines what the page is about in one line.',
   483→        'Add one clear H1 near the top stating your primary offer.',
   484→        'Relevance clarity',
   485→        'quick'
   486→      )
   487→    );
   488→
   489→  checks.push({
   490→    id: 'heading-structure',
   491→    label: 'Subheading structure (H2s)',
   492→    status: p.h2s.length >= 2 ? 'pass' : p.h2s.length === 1 ? 'warn' : 'warn',
   493→    detail: `${p.h2s.length} H2, ${p.h3s.length} H3`,
   494→  });
   495→  if (p.h2s.length < 2)
   496→    findings.push(
   497→      finding(
   498→        'few-h2',
   499→        'content',
   500→        'low',
   501→        'Very few subheadings on the homepage',
   502→        [`${p.h2s.length} H2 heading(s) found`],
   503→        'Subheadings organize the page for readers and give search engines context about your services.',
   504→        'Break the homepage into scannable sections with descriptive H2s (services, areas served, why us, proof).',
   505→        'Scannability and topical relevance',
   506→        'medium'
   507→      )
   508→    );
   509→
   510→  const withAlt = p.images.filter((i) => (i.alt ?? '').length > 0).length;
   511→  const altPct = p.images.length ? Math.round((withAlt / p.images.length) * 100) : 100;
   512→  checks.push({
   513→    id: 'alt-coverage',
   514→    label: 'Image alt text coverage',
   515→    status: p.images.length === 0 ? 'na' : altPct >= 90 ? 'pass' : altPct >= 50 ? 'warn' : 'fail',
   516→    detail: p.images.length === 0 ? 'No images found' : `${withAlt}/${p.images.length} images have alt text (${altPct}%)`,
   517→  });
   518→  if (p.images.length > 0 && altPct < 90)
   519→    findings.push(
   520→      finding(
   521→        'alt-text',
   522→        'content',
   523→        altPct < 50 ? 'medium' : 'low',
   524→        `${p.images.length - withAlt} images missing alt text`,
   525→        [
   526→          `${withAlt} of ${p.images.length} images have alt text (${altPct}%)`,
   527→          ...p.images.filter((i) => !i.alt).slice(0, 3).map((i) => i.src.split('/').slice(-1)[0].slice(0, 60)),
   528→        ],
   529→        'Alt text makes images accessible to screen readers and tells search engines what pictures show. It is also required for image search visibility.',
   530→        'Describe each meaningful image in a short alt attribute (what it shows, in context).',
   531→        'Accessibility and image search',
   532→        'quick'
   533→      )
   534→    );
   535→
   536→  checks.push({
   537→    id: 'internal-links',
   538→    label: 'Internal linking',
   539→    status: p.internalLinkCount >= 5 ? 'pass' : p.internalLinkCount >= 1 ? 'warn' : 'fail',
   540→    detail: `${p.internalLinkCount} internal links on homepage`,
   541→  });
   542→  if (p.internalLinkCount < 5)
   543→    findings.push(
   544→      finding(
   545→        'internal-links',
   546→        'content',
   547→        'medium',
   548→        'Very few internal links',
   549→        [`${p.internalLinkCount} internal links found on the homepage`],
   550→        'Internal links spread authority to your service and location pages and guide visitors to the next step.',
   551→        'Link prominently to your main service pages, location pages, and contact page from the homepage.',
   552→        'Page authority distribution',
   553→        'medium'
   554→      )
   555→    );
   556→
   557→  checks.push({
   558→    id: 'home-word-count',
   559→    label: 'Homepage content depth',
   560→    status: p.wordCount >= 400 ? 'pass' : p.wordCount >= 150 ? 'warn' : 'fail',
   561→    detail: `${p.wordCount} words of visible text`,
   562→  });

  1056→    findings.push(
  1057→      finding(
  1058→        'no-lazy-loading',
  1059→        'performance',
  1060→        'medium',
  1061→        `${noLazy.length} images load eagerly`,
  1062→        [`${noLazy.length} of ${p.images.length} images lack loading="lazy"`],
  1063→        'Browsers download non-lazy images immediately, even ones far below the fold. On image-heavy sites this is often the single biggest fix.',
  1064→        'Add loading="lazy" to images below the fold (most CMSs can do this site-wide).',
  1065→        'Initial load time',
  1066→        'quick'
  1067→      )
  1068→    );
  1069→
  1070→  const noDims = p.images.filter((i) => !i.hasDimensions);
  1071→  checks.push({
  1072→    id: 'image-dimensions',
  1073→    label: 'Image dimensions set',
  1074→    status: p.images.length === 0 ? 'na' : noDims.length / p.images.length < 0.3 ? 'pass' : 'warn',
  1075→    detail: `${noDims.length} of ${p.images.length} images missing width/height`,
  1076→  });
  1077→
  1078→  const compression = ctx.headers['content-encoding'] ?? null;
  1079→  checks.push({
  1080→    id: 'perf-compression',
  1081→    label: 'Compression enabled',
  1082→    status: compression ? 'pass' : 'warn',
  1083→    detail: compression ?? 'No Content-Encoding on HTML',
  1084→  });
  1085→
  1086→  checks.push({
  1087→    id: 'cache-headers',
  1088→    label: 'Cache headers',
  1089→    status: ctx.headers['cache-control'] ? 'pass' : 'warn',
  1090→    detail: ctx.headers['cache-control'] ?? 'No Cache-Control header on HTML response',
  1091→  });
  1092→
  1093→  return { key: 'performance', label: 'Performance', checks, findings };
  1094→}

  1099→export function computeCategoryScores(outputs: AnalyzerOutput[]): CategoryScore[] {
  1100→  return outputs.map((o) => {
  1101→    const counted = o.checks.filter((c) => c.status !== 'na');
  1102→    const points = counted.reduce((n, c) => n + (c.status === 'pass' ? 1 : c.status === 'warn' ? 0.5 : 0), 0);
  1103→    const score = counted.length ? Math.round((points / counted.length) * 100) : 0;
  1104→    return { key: o.key, label: o.label, score, checks: o.checks };
  1105→  });
  1106→}

  1108→export function computeOverall(scores: CategoryScore[]): number {
  1109→  const byKey = Object.fromEntries(scores.map((s) => [s.key, s.score]));
  1110→  const localNa = scores.find((s) => s.key === 'local');
  1111→  const localIsWeakData = localNa && localNa.checks.filter((c) => c.status === 'na').length >= 3;
  1112→  const weights = localIsWeakData
  1113→    ? { technical: 0.28, content: 0.28, ai: 0.23, performance: 0.21 }
  1114→    : { technical: 0.24, content: 0.24, ai: 0.2, performance: 0.18, local: 0.14 };
  1115→  let total = 0;
  1116→  let weightSum = 0;
  1117→  for (const [key, w] of Object.entries(weights)) {
  1118→    const v = (byKey as Record<string, number>)[key];
  1119→    if (typeof v === 'number') {
  1120→      total += v * w;
  1121→      weightSum += w;
  1122→    }
  1123→  }
  1124→  return weightSum ? Math.round(total / weightSum) : 0;
  1125→}
